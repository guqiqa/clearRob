cmake_minimum_required(VERSION 3.8)
project(remotecontrol_wheel_pid_v2_navspeed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(navigation_interfaces REQUIRED)
add_executable(RemoteControl_WheelPID_V2 main.cpp odometry.cpp)
target_compile_definitions(RemoteControl_WheelPID_V2 PRIVATE REMOTECONTROL_HAS_ROS2)
ament_target_dependencies(RemoteControl_WheelPID_V2
  rclcpp geometry_msgs nav_msgs std_srvs tf2_ros navigation_interfaces)
install(TARGETS RemoteControl_WheelPID_V2 DESTINATION lib/${PROJECT_NAME})
ament_package()
